const button = cva('rounded-md font-medium', {
  variants: {
    variant: {
      primary: 'bg-[--primary] text-white',
      secondary: 'bg-muted border',
      ghost: 'bg-transparent',
    },
    size: { sm: 'h-8 px-3', md: 'h-10 px-4' },

});

  return <button className={button({ variant, size })} {...props} />;
